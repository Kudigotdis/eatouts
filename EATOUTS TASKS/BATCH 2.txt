  if(id==='area'){d.area=val;var t=DL()&&DL().tierFor?DL().tierFor(val):null;d.tier=t?t.id:null}
  else if(id==='propType'){d.propType=val;d.cond={}}
  else if(id==='gender'){STATE.recipient.gender=val}
  else if(id==='mobileCC'){STATE.recipient.country=val}
  else if(id==='bank'){STATE.payment.bank=val}
}

/* ---------- events ---------- */
document.addEventListener('click',function(e){
  if(e.target.classList&&e.target.classList.contains('sheet-scrim')){if(STATE.infoPopup){STATE.infoPopup=null;render();return}if(STATE.photoPopup){STATE.photoPopup=null;STATE.editRcpt=null;render();return}}
  if(!STATE.openDrp)return;
  if(e.target.closest('.drp-opt'))return;
  if(e.target.closest('[data-drp-open]'))return;
  if(!e.target.closest('.drp-list')){STATE.openDrp=null;render()}
});
document.addEventListener('keydown',function(e){if(e.key!=='Escape')return;if(STATE.infoPopup||STATE.photoPopup||STATE.openDrp){STATE.infoPopup=null;STATE.photoPopup=null;STATE.editRcpt=null;STATE.openDrp=null;render()}});
