**Insert immediately after the closing `}` of `cartKeys()`:**

```js
/* D10 (option A): the cart persists under 'eatouts_cart_v1' so a promo
   added on venue-about.html survives a reload and reappears on the menu
   badge. Cart keys and cart logic stay in this file (DO-NOT: no move).
   Promo lines use the namespaced key "promo:<slug>:<promoId>" so
   cartKeys() — which filters on "i<restId>:" — never returns them and
   menu totals / checkout stay menu-only. */
const CART_KEY="eatouts_cart_v1";
function cartPersist(){
  try{
    localStorage.setItem(CART_KEY,JSON.stringify({
      v:1,cart:state.cart,updated:new Date().toISOString()
    }));
  }catch(e){}
}
function cartHydrate(){
  try{
    const raw=localStorage.getItem(CART_KEY);
    if(!raw) return false;
    const p=JSON.parse(raw);
    if(!p||p.v!==1||!p.cart||typeof p.cart!=="object") return false;
    state.cart=p.cart;
    return true;
  }catch(e){return false;}
}
/* promo:* only — menu totals stay covered by cartTotal() */
function cartPromoTotal(){
  let t=0;
  Object.keys(state.cart).forEach(function(k){
    if(k.indexOf("promo:")!==0)return;
    const l=state.cart[k];if(!l)return;
    t+=(Number(l.price)||0)*(Number(l.qty)||0);
  });
  return t;
}
function cartPromoCount(){
  let c=0;
  Object.keys(state.cart).forEach(function(k){
    if(k.indexOf("promo:")!==0)return;
    const l=state.cart[k];if(!l)return;
    c+=(Number(l.qty)||0);
  });
  return c;
}
/* Public write path. One row per (slug,promoId); tapping again
   increments qty on the same row (never duplicates). */
window.EO_cart={
  addPromo:function(line){
    if(!line||!line.venueSlug||!line.promoId) return false;
    const key="promo:"+String(line.venueSlug)+":"+String(line.promoId);
    const cur=state.cart[key];
    if(cur){
      cur.qty=(Number(cur.qty)||0)+1;
      if(line.title) cur.title=line.title;
      if(line.price!=null) cur.price=Number(line.price);
    }else{
      state.cart[key]={
        itemId:key,sizeIdx:-1,qty:1,extras:{},kind:"promo",
        title:line.title||"Promo",price:Number(line.price)||0
      };
    }
    cartPersist();
    return true;
  },
  total:function(){return cartTotal()+cartPromoTotal();},
  keys:function(){return Object.keys(state.cart);}
};
```
