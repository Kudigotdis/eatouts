+  /* ------------------------------------------------------------
+     Cart persistence (D10 option A) — writes on every mutation,
+     hydrates on boot, mirrors across tabs via `storage`.
+     Shared key with venue-about.html ("Add to Order").
+     ------------------------------------------------------------ */
+  var CART_KEY = 'eatouts_cart_v1';
+  function cartRead(){
+    try { var raw = localStorage.getItem(CART_KEY); return raw ? JSON.parse(raw) : null; }
+    catch (e) { return null; }
+  }
+  function cartWrite(){
+    try { localStorage.setItem(CART_KEY, JSON.stringify(state.cart || {})); }
+    catch (e) { /* quota / disabled — memory state stays authoritative */ }
+  }
+  function cartHydrate(){
+    var saved = cartRead();
+    if (saved && typeof saved === 'object') state.cart = saved;
+  }
+  /* Hydrate before first render so totals/badges are correct on load. */
+  cartHydrate();
+  /* Pick up cross-tab writes (venue-about "Add to Order", other SPA tab). */
+  window.addEventListener('storage', function (e) {
+    if (!e || e.key !== CART_KEY) return;
+    var next = cartRead();
+    if (next && typeof next === 'object') {
+      state.cart = next;
+      try { if (typeof syncCart === 'function') syncCart(); } catch (_) {}
+      try { if (typeof render === 'function')    render();   } catch (_) {}
+    }
+  });

### 4d — `venue-about.html` — JS (insert near the promo carousel block)

```diff
+/* ------------------------------------------------------------
+   T09 — promo lightbox + "Add to Order" (D10 option A)
+   Persists to `eatouts_cart_v1` — same key index.html hydrates.
+   ------------------------------------------------------------ */
+var PLB_INDEX = 0;
+var PLB_IMAGES = [];
+var PLB_PROMO = null;   // { id, title, price }
+var PLB_NODE = null;
+var PLB_CART_KEY = 'eatouts_cart_v1';
+
+function plbReadCart(){
+  try { var raw = localStorage.getItem(PLB_CART_KEY); return raw ? JSON.parse(raw) : {}; }
+  catch (e) { return {}; }
+}
+function plbWriteCart(cart){
+  try { localStorage.setItem(PLB_CART_KEY, JSON.stringify(cart || {})); } catch (e) {}
+}
+function plbToast(msg){
+  var t = document.querySelector('.plb-toast');
+  if (!t) { t = document.createElement('div'); t.className = 'plb-toast'; document.body.appendChild(t); }
+  t.textContent = msg;
+  t.classList.add('on');
+  clearTimeout(plbToast._t);
+  plbToast._t = setTimeout(function(){ t.classList.remove('on'); }, 1800);
+}
+function plbPaint(){
+  if (!PLB_NODE) return;
+  var img = PLB_NODE.querySelector('.plb-img');
+  var dots = PLB_NODE.querySelector('.plb-dots');
+  if (img)  img.src = PLB_IMAGES[PLB_INDEX] || '';
+  if (dots) dots.innerHTML = PLB_IMAGES.map(function (_, i) {
+    return '<span class="plb-dot' + (i === PLB_INDEX ? ' on' : '') + '"></span>';
+  }).join('');
+}
+function plbClose(){
+  if (PLB_NODE && PLB_NODE.parentNode) PLB_NODE.parentNode.removeChild(PLB_NODE);
+  PLB_NODE = null;
+  document.removeEventListener('keydown', plbKeydown);
+}
+function plbKeydown(e){
+  if (!e) return;
+  if (e.key === 'Escape' || e.keyCode === 27) return plbClose();
+  if (e.key === 'ArrowRight' && PLB_INDEX < PLB_IMAGES.length - 1) { PLB_INDEX++; plbPaint(); }
+  if (e.key === 'ArrowLeft'  && PLB_INDEX > 0)                    { PLB_INDEX--; plbPaint(); }
+}
+function plbAddToOrder(){
+  if (!PLB_PROMO) { plbToast('Promo not available'); return; }
+  var id    = String(PLB_PROMO.id    || '').trim() || ('promo_' + Date.now());
+  var title = String(PLB_PROMO.title || 'Promo');
+  var price = Number(PLB_PROMO.price != null ? PLB_PROMO.price : 0) || 0;
+  var cart  = plbReadCart();
+  if (cart[id]) cart[id].qty = (Number(cart[id].qty) || 1) + 1;
+  else          cart[id] = { id: id, title: title, price: price, qty: 1, addedAt: Date.now() };
+  plbWriteCart(cart);
+  plbToast('Added to order');
+  plbClose();
+}
+function plbOpen(promo, images){
+  if (!images || !images.length) return;
+  PLB_PROMO = promo || null;
+  PLB_IMAGES = images.slice();
+  PLB_INDEX = 0;
+  var scrim = document.createElement('div');
+  scrim.className = 'plb-scrim';
+  scrim.setAttribute('role', 'dialog');
+  scrim.setAttribute('aria-modal', 'true');
+  scrim.setAttribute('aria-label', 'Promo image');
+  scrim.innerHTML =
+    '<div class="plb-box">' +
+      '<div class="plb-track"><img class="plb-img" alt=""></div>' +
+      '<div class="plb-dots" aria-hidden="true"></div>' +
+      '<button type="button" class="plb-add">Add to Order</button>' +
+    '</div>';
+  document.body.appendChild(scrim);
+  PLB_NODE = scrim;
+  plbPaint();
+
+  /* swipe */
+  var x0 = 0, y0 = 0;
+  var track = scrim.querySelector('.plb-track');
+  track.addEventListener('touchstart', function (e) {
+    if (e.touches && e.touches[0]) { x0 = e.touches[0].clientX; y0 = e.touches[0].clientY; }
+  }, { passive: true });
+  track.addEventListener('touchend', function (e) {
+    if (!e.changedTouches || !e.changedTouches[0]) return;
+    var dx = e.changedTouches[0].clientX - x0;
+    var dy = e.changedTouches[0].clientY - y0;
+    if (Math.abs(dx) < 30 || Math.abs(dx) < Math.abs(dy)) return;
+    if (dx < 0) PLB_INDEX = Math.min(PLB_IMAGES.length - 1, PLB_INDEX + 1);
+    else        PLB_INDEX = Math.max(0, PLB_INDEX - 1);
+    plbPaint();
+  }, { passive: true });
+
+  /* outside tap */
+  scrim.addEventListener('click', function (e) { if (e.target === scrim) plbClose(); });
+  document.addEventListener('keydown', plbKeydown);
+  scrim.querySelector('.plb-add').addEventListener('click', plbAddToOrder);
+}
+
+/* Delegated click on the promo thumb */
+document.addEventListener('click', function (e) {
+  var t = e.target.closest && e.target.closest('.promo-thumb');
+  if (!t) return;
+  e.preventDefault();
+  var promo = {
+    id:    t.getAttribute('data-promo-id'),
+    title: t.getAttribute('data-promo-title'),
+    price: parseFloat(t.getAttribute('data-promo-price')) || 0
+  };
+  var images = [];
+  try { images = JSON.parse(t.getAttribute('data-promo-images') || '[]'); } catch (_) {}
+  if (!images.length) images = Array.prototype.map.call(
+    t.querySelectorAll('img'), function (i) { return i.getAttribute('src'); }
+  ).filter(Boolean);
+  plbOpen(promo, images);
+});
```
